```python
def create_model_family_encoder_column(df):
    """
    Create composite key combining model_family and encoder_config.
    RAFT has no encoder variants, CatsPP has 4 (FF/FT/TF/TT).
    """
    if 'model_family' not in df.columns:
        return df
    
    df = df.copy()
    
    def make_key(row):
        family = str(row.get('model_family', '')).lower()
        if family == 'raft':
            return 'raft'
        encoder = str(row.get('encoder_config', ''))
        if encoder and family:
            return f"{family}_{encoder}"
        return family if family else encoder
    
    df['model_family_encoder'] = df.apply(make_key, axis=1)
    return df
```

```python
def compute_univariate_predictor_comparison(
    df,
    predictors,
    target,
    output_path=None,
    standardize=True,
    ridge_alpha=1.0,
    cv_standardize_mode="global",
    target_demean_groups=None,
):
    """
    Test each predictor individually to determine standalone predictive power.
    
    Returns DataFrame with columns:
    - predictor: name
    - n: sample size
    - r2: in-sample R²
    - lobo_r2: LOBO R² (mean across benchmarks)
    - lobo_pearson: LOBO Pearson correlation
    - lobo_spearman: LOBO Spearman correlation
    - lobo_mae: mean absolute error
    - lobo_rmse: root mean squared error
    """
    complete_df = filter_complete_rows(df, predictors, target)
    if complete_df.empty:
        return pd.DataFrame()
    
    results = []
    
    for pred in predictors:
        if pred not in complete_df.columns:
            continue
        
        # In-sample R²
        X = complete_df[[pred]].values
        y = complete_df[target].values
        
        if standardize:
            X_std = (X - X.mean(axis=0)) / (X.std(axis=0) + 1e-10)
        else:
            X_std = X
        
        X_with_intercept = np.column_stack([np.ones(len(X_std)), X_std])
        penalty = np.eye(2)
        penalty[0, 0] = 0.0
        coef = np.linalg.solve(
            X_with_intercept.T @ X_with_intercept + float(ridge_alpha) * penalty,
            X_with_intercept.T @ y
        )
        
        y_pred = X_with_intercept @ coef
        ss_res = np.sum((y - y_pred) ** 2)
        ss_tot = np.sum((y - np.mean(y)) ** 2)
        r2 = 1.0 - (ss_res / ss_tot) if ss_tot != 0 else 0.0
        
        # LOBO validation
        lobo_summary, _ = run_group_cv(
            complete_df,
            "benchmark",
            [pred],
            target,
            standardize=standardize,
            standardize_mode=cv_standardize_mode,
            model="ridge",
            ridge_alpha=ridge_alpha,
            target_group_demean=len(target_demean_groups) > 0 if target_demean_groups else False,
            target_group_col=target_demean_groups,
        )
        
        if not lobo_summary.empty:
            overall = lobo_summary[lobo_summary["benchmark"] == "__overall__"]
            if not overall.empty:
                lobo_r2 = float(overall["r2"].iloc[0])
                lobo_pearson = float(overall["pearson"].iloc[0])
                lobo_spearman = float(overall["spearman"].iloc[0])
                lobo_mae = float(overall["mae"].iloc[0])
                lobo_rmse = float(overall["rmse"].iloc[0])
            else:
                lobo_r2 = lobo_pearson = lobo_spearman = lobo_mae = lobo_rmse = np.nan
        else:
            lobo_r2 = lobo_pearson = lobo_spearman = lobo_mae = lobo_rmse = np.nan
        
        results.append({
            'predictor': pred,
            'n': len(complete_df),
            'r2': r2,
            'lobo_r2': lobo_r2,
            'lobo_pearson': lobo_pearson,
            'lobo_spearman': lobo_spearman,
            'lobo_mae': lobo_mae,
            'lobo_rmse': lobo_rmse,
        })
    
    result_df = pd.DataFrame(results).sort_values('lobo_r2', ascending=False)
    
    if output_path:
        result_df.to_csv(output_path, index=False)
        print(f"Univariate predictor comparison saved to {output_path}")
    
    return result_df
```
